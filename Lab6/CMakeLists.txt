cmake_minimum_required(VERSION 3.15)
project(lab06 CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_executable(lab06 WIN32 main.cpp)
target_link_libraries(lab06 user32 gdi32 winspool ole32 oleaut32)
if(MINGW)
  target_link_options(lab06 PRIVATE -municode)
endif()
