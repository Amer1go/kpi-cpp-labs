cmake_minimum_required(VERSION 3.15)
project(lab05 CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_executable(lab05 WIN32 main.cpp)
target_link_libraries(lab05 user32 gdi32)
if(MINGW)
  target_link_options(lab05 PRIVATE -municode)
endif()
